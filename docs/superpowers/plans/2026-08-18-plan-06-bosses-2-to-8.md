```js
  // Per-stage stats
  const stats = [
    { r: 50, hp:  800, phaseCount: 3, spawnMinions: false },  // stage 1

    { r: 50, hp: 1100, phaseCount: 3, spawnMinions: false },  // stage 3

    { r: 50, hp: 1400, phaseCount: 4, spawnMinions: false },  // stage 7
    { r: 75, hp: 2000, phaseCount: 5, spawnMinions: true  },  // stage 8
  ];
  const s = stats[stageNum - 1];
  bossMaxHp = s.hp;
  bossPhase = 0;
  bossTimer = 0;
  bossAngle = 0;
  return {
    stageNum,
    x: W/2, y: 130,
    r: s.r,
    hp: s.hp,
    targetX: W/2, targetY: 130,
    spd: 58,
    fireTimer: 1.8,
    phaseCount:    s.phaseCount,
    spawnMinions:  s.spawnMinions,
    minionTimer:   3.0,
    phantomAlpha:  1.0,  // stage 7 boss transparency
  };
}
```

```js
    const rx = Math.cos(a) * outerR * distort;
    const ry = Math.sin(a) * outerR * distort;
    a === 0 ? ctx.moveTo(rx, ry) : ctx.lineTo(rx, ry);
  }
  ctx.closePath();
  ctx.fillStyle = '#550010'; ctx.fill();
  ctx.strokeStyle = '#bb0022'; ctx.lineWidth = 2; ctx.stroke();

  ctx.restore();
```

```js
function fireBoss() {
  if (!player || player.dead || !boss) return;
  switch (boss.stageNum) {
    case 1: fireBoss1(); break;
    case 2: fireBoss2(); break;

function fireBoss1() {
  // Original 3-phase patterns
  const dx = player.x - boss.x, dy = player.y - boss.y;
  const d   = Math.sqrt(dx*dx + dy*dy) || 1;
  const spd = (175 + bossPhase * 35) * diffMult;
  const mkEB = (vx, vy, clr) =>
    enemyBullets.push({ x: boss.x, y: boss.y, vx, vy, r: 5, clr });
  switch (bossPhase) {
    case 0: // wide spread
      for (let i = -3; i <= 3; i++) {
        const ang = Math.atan2(dy, dx) + i * 0.14;
        mkEB(Math.cos(ang)*spd, Math.sin(ang)*spd, '#ff2200');
      }
      break;
    case 1: // aimed twin streams
      [-0.08, 0.08].forEach(a => {
        const ang = Math.atan2(dy, dx) + a;
        mkEB(Math.cos(ang)*spd, Math.sin(ang)*spd, '#ff8800');
      });
      break;
    case 2: // rotating ring
      for (let i = 0; i < 8; i++) {
        const ang = bossAngle + i * Math.PI*2/8;
        mkEB(Math.cos(ang)*spd*0.7, Math.sin(ang)*spd*0.7, '#cc00ff');
      }
      break;
  }
}

function fireBoss3() {

function fireBoss5() {
  // Solar Core: sweeping arc of bullets (rotates with bossAngle) + aimed
  const dx = player.x - boss.x, dy = player.y - boss.y;

  const d   = Math.sqrt(dx*dx + dy*dy) || 1;
  const spd = (175 + bossPhase * 28) * diffMult;
  const mkEB = (vx, vy, clr, r) =>
    enemyBullets.push({ x: boss.x, y: boss.y, vx, vy, r: r||5, clr });
  // Ring component (always active)
  const ringCount = 8 + bossPhase * 2;  // 8, 10, 12, 14, 16 bullets
  for (let i = 0; i < ringCount; i++) {
    const a = bossAngle + (i / ringCount) * Math.PI * 2;
    mkEB(Math.cos(a)*spd*0.65, Math.sin(a)*spd*0.65, '#ff2200');
  }
  // Aimed component (phase 1+)
  if (bossPhase >= 1) {
    const baseA = Math.atan2(dy, dx);
    [-0.10, 0, 0.10].forEach(off => {
      mkEB(Math.cos(baseA+off)*spd, Math.sin(baseA+off)*spd, '#ff8800');
    });
  }
  // Scatter component (phase 2+)
  if (bossPhase >= 2) {
    for (let i = 0; i < 4; i++) {
      const a = Math.random() * Math.PI * 2;
      mkEB(Math.cos(a)*spd*0.75, Math.sin(a)*spd*0.75, '#ffaa00');
    }
  }
}
```

## Step 4 — Refactor `updateBoss(dt)` for per-boss phase count, phantom alpha, and minion spawning

```js
function updateBoss(dt) {
  if (!boss) return;
  bossTimer += dt;
  bossAngle += dt * 0.85;

  // Patrol movement
  const dx = boss.targetX - boss.x, dy = boss.targetY - boss.y;
  const d  = Math.sqrt(dx*dx + dy*dy) || 1;
  if (d > 5) {
    boss.x += dx/d * boss.spd * dt;
    boss.y += dy/d * boss.spd * dt;
  } else {
    boss.targetX = 80 + Math.random() * (W - 160);
    boss.targetY = 60 + Math.random() * 140;
  }

  // Phase from HP percentage — uses per-boss phaseCount
  const hpPct = boss.hp / bossMaxHp;
  // Evenly divide HP into phaseCount buckets (phase 0 = full HP, phase N-1 = almost dead)
  bossPhase = boss.phaseCount - 1 - Math.floor(hpPct * boss.phaseCount);
  bossPhase = Math.max(0, Math.min(boss.phaseCount - 1, bossPhase));

  // Boss 7: phantom alpha oscillation
  if (boss.stageNum === 7) {
    boss.phantomAlpha = 0.65 + Math.sin(bossTimer * 1.5) * 0.35;
  }

  // Minion spawning (stages 6 and 8)
  if (boss.spawnMinions) {
    boss.minionTimer -= dt;
    if (boss.minionTimer <= 0) {
      spawnMinion();
      boss.minionTimer = 3.0 / diffMult;  // faster in higher difficulty
    }
  }

  // Fire
  boss.fireTimer -= dt;
  if (boss.fireTimer <= 0) {
    fireBoss();
    const rate = [1.2, 0.85, 0.55, 0.38, 0.28][Math.min(bossPhase, 4)];
    boss.fireTimer = (rate / diffMult) + Math.random() * 0.25;
  }
}
```
